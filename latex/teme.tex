\section{Pn cu metoda analizei combinatorii a lui Euler}
\[
\begin{aligned}
P_{n+2} &= 2P_{n+1}+P_n, \qquad n\ge 0,\\
P_0 &= 0, \qquad P_1=1,\\
P(x) &= \sum_{n=0}^{\infty} P_n x^n
      = \frac{x}{1-2x-x^2}.
\end{aligned}
\]
